using System;
using System.Collections.Generic;

public class HelloWorld
{
    public static void Main(string[] args)
    {
        int n = int.Parse(Console.ReadLine());

        Dictionary<string, decimal> spendByCategory 
            = new Dictionary<string, decimal>();

        for(int i=0;i<n;i++)
        {
            string s = Console.ReadLine();
            string[] arr = s.Split(' ', StringSplitOptions.RemoveEmptyEntries);

            string type = arr[0];
            decimal exp = decimal.Parse(arr[1]);

            if(exp < 0)
            {
                decimal val = Math.Abs(exp);

                if(spendByCategory.ContainsKey(type))
                    spendByCategory[type] += val;
                else
                    spendByCategory[type] = val;
            }
        }

        foreach(var item in spendByCategory)
            Console.WriteLine(item.Key + ":" + item.Value);
    }
}
